# Create one lib/ui archive for one panel build.
#
# lib/ui compiles the 320x240 or 480x320 layout, fonts and precomputed assets
# according to PIPOD_DISPLAY_ST7796 (see ui_layout.h / ui_assets_build.c), so
# each firmware needs its own archive. The default `ui` target is the ILI9341
# build; apps/pipod creates `ui_st7796` with pipod_display_select().
set(PIPOD_UI_DIR "${CMAKE_CURRENT_LIST_DIR}" CACHE INTERNAL "lib/ui source directory")
function(pipod_add_ui_library target gfx_target)
    add_library(${target} STATIC
        ${PIPOD_UI_DIR}/ui_core.c
        ${PIPOD_UI_DIR}/ui_shared.c
        ${PIPOD_UI_DIR}/ui_anim.c
        ${PIPOD_UI_DIR}/ui_widgets.c
        ${PIPOD_UI_DIR}/ui_menu.c
        ${PIPOD_UI_DIR}/ui_nowplaying.c
        ${PIPOD_UI_DIR}/ui_settings.c
    )

    # Precomputed assets for the flash -> panel direct layer path (generated by
    # scripts/gen_ui_assets.py; `const` arrays -> flash). ui_assets_build.c
    # includes the 320x240 or 480x320 set according to ui_layout.h.
    target_sources(${target} PRIVATE
        ${PIPOD_UI_DIR}/assets/ui_assets_build.c
    )

    # ui.h is the public contract; consumers include it as "ui.h".
    target_include_directories(${target} PUBLIC
        ${PIPOD_UI_DIR}
        ${PIPOD_UI_DIR}/assets
    )

    target_link_libraries(${target} PUBLIC
        ${gfx_target}
        ili9341
        st7796
        pico_stdlib
        pico_multicore
        pico_util
        hardware_sync
    )

    # Rendering is hot: it shares core 1 with input handling at 30 FPS.
    target_compile_options(${target} PRIVATE -O2)
endfunction()

pipod_add_ui_library(ui gfx)
