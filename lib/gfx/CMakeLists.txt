# Create one lib/gfx archive for one panel build.
#
# The renderer's strip buffers and every dirty-rect clamp are sized from
# PIPOD_DISPLAY_ST7796 (gfx_config.h), so the ILI9341 and ST7796 firmwares need
# separate archives built from the same sources. The default `gfx` target is the
# ILI9341 build; apps/pipod creates `gfx_st7796` with pipod_display_select().
set(PIPOD_GFX_DIR "${CMAKE_CURRENT_LIST_DIR}" CACHE INTERNAL "lib/gfx source directory")
function(pipod_add_gfx_library target)
    add_library(${target} STATIC
        ${PIPOD_GFX_DIR}/gfx.c
        ${PIPOD_GFX_DIR}/gfx_render.c
    )

    target_include_directories(${target} PUBLIC
        ${PIPOD_GFX_DIR}
        ${PIPOD_GFX_DIR}/fonts
    )

    # display.h resolves to one driver at compile time (PIPOD_DISPLAY_ST7796),
    # so both archives are linked and only the referenced one is pulled in.
    target_link_libraries(${target} PUBLIC
        ili9341
        st7796
        pico_stdlib
    )

    # Drawing is hot: the UI renders strips on core 1 while core 0 runs audio.
    target_compile_options(${target} PRIVATE -O2)

    # Generated fonts (owned by the asset pipeline). Globbing the directory
    # instead of including fonts/fonts.cmake keeps the helper usable for more
    # than one target; fonts.cmake is still regenerated by scripts.
    file(GLOB _gfx_font_sources CONFIGURE_DEPENDS
        ${PIPOD_GFX_DIR}/fonts/*.c)
    target_sources(${target} PRIVATE ${_gfx_font_sources})
endfunction()

pipod_add_gfx_library(gfx)
